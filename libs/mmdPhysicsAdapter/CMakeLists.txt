cmake_minimum_required(VERSION 3.22)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/UsdMmdProject.cmake")
project(mmdPhysicsAdapter
    VERSION ${USDMMD_VERSION}
    DESCRIPTION "MMD bone/body coupling over composed USD declarations"
    LANGUAGES CXX)
usdmmd_component(TESTS_OPTION MMDPHYSICSADAPTER_BUILD_TESTS)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/UsdMmdTargets.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/UsdMmdPackage.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/UsdMmdOpenUsd.cmake")
usdmmd_find_openusd()
if(NOT TARGET mmdControl::mmdControl)
    find_package(mmdControl CONFIG REQUIRED)
endif()
add_library(mmdPhysicsAdapter STATIC src/Adapter.cpp src/BodyProperties.cpp src/JointSprings.cpp)
add_library(mmdPhysicsAdapter::mmdPhysicsAdapter ALIAS mmdPhysicsAdapter)
set_target_properties(mmdPhysicsAdapter PROPERTIES EXPORT_NAME mmdPhysicsAdapter POSITION_INDEPENDENT_CODE ON)
target_compile_features(mmdPhysicsAdapter PUBLIC cxx_std_20)
target_include_directories(mmdPhysicsAdapter PUBLIC
    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
    "$<INSTALL_INTERFACE:include>")
target_link_libraries(mmdPhysicsAdapter PUBLIC mmdControl::mmdControl)
if(TARGET usd_ms)
    target_link_libraries(mmdPhysicsAdapter PUBLIC usd_ms)
else()
    foreach(_t IN ITEMS tf gf vt sdf usd usdGeom usdSkel usdPhysics)
        if(TARGET ${_t})
            target_link_libraries(mmdPhysicsAdapter PUBLIC ${_t})
        elseif(TARGET pxr::${_t})
            target_link_libraries(mmdPhysicsAdapter PUBLIC pxr::${_t})
        else()
            message(FATAL_ERROR "Missing OpenUSD target ${_t}")
        endif()
    endforeach()
endif()
usdmmd_target_defaults(mmdPhysicsAdapter)
usdmmd_install_library(mmdPhysicsAdapter)
if(MMDPHYSICSADAPTER_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
