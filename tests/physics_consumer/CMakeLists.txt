# Opt-in contract probe, separate from the importer and normal workspace build.
cmake_minimum_required(VERSION 3.25)
project(MmdPhysicsConsumerProbe LANGUAGES CXX)

find_package(physicsUsd 0.1 CONFIG REQUIRED)
add_executable(mmd_physics_consumer_probe main.cpp)
target_compile_features(mmd_physics_consumer_probe PRIVATE cxx_std_17)
target_link_libraries(mmd_physics_consumer_probe PRIVATE physicsUsd::physicsUsd)
option(MMD_PHYSICS_EXECUTION "Exercise the installed optional six-DOF backend" OFF)
if(MMD_PHYSICS_EXECUTION)
  find_package(physicsJolt 0.1 CONFIG REQUIRED)
  if(NOT physicsJolt_SIX_DOF_AVAILABLE)
    message(FATAL_ERROR "Execution probe requires physicsJolt built with PHYSICSJOLT_ENABLE_SIX_DOF=ON")
  endif()
  target_link_libraries(mmd_physics_consumer_probe PRIVATE physicsJolt::physicsJolt)
  target_compile_definitions(mmd_physics_consumer_probe PRIVATE MMD_PHYSICS_EXECUTION)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(mmd_physics_filter_test collision_filter_test.cpp)
  target_compile_features(mmd_physics_filter_test PRIVATE cxx_std_17)
  target_link_libraries(mmd_physics_filter_test PRIVATE physicsUsd::physicsUsd)
  if(MMD_PHYSICS_EXECUTION)
    target_link_libraries(mmd_physics_filter_test PRIVATE physicsJolt::physicsJolt)
    target_compile_definitions(mmd_physics_filter_test PRIVATE MMD_PHYSICS_EXECUTION)
  endif()
  add_test(NAME mmd_physics_collision_filter COMMAND mmd_physics_filter_test)
  add_executable(mmd_physics_material_test contact_material_test.cpp)
  target_compile_features(mmd_physics_material_test PRIVATE cxx_std_17)
  target_link_libraries(mmd_physics_material_test PRIVATE physicsUsd::physicsUsd)
  if(MMD_PHYSICS_EXECUTION)
    target_link_libraries(mmd_physics_material_test PRIVATE physicsJolt::physicsJolt)
    target_compile_definitions(mmd_physics_material_test PRIVATE MMD_PHYSICS_EXECUTION)
  endif()
  add_test(NAME mmd_physics_contact_material COMMAND mmd_physics_material_test)
  add_executable(mmd_physics_damping_test body_damping_test.cpp)
  target_compile_features(mmd_physics_damping_test PRIVATE cxx_std_17)
  target_link_libraries(mmd_physics_damping_test PRIVATE physicsUsd::physicsUsd)
  if(MMD_PHYSICS_EXECUTION)
    target_link_libraries(mmd_physics_damping_test PRIVATE physicsJolt::physicsJolt)
    target_compile_definitions(mmd_physics_damping_test PRIVATE MMD_PHYSICS_EXECUTION)
  endif()
  add_test(NAME mmd_physics_body_damping COMMAND mmd_physics_damping_test)
endif()
