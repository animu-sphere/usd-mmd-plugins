# Opt-in contract probe, separate from the importer and normal workspace build.
cmake_minimum_required(VERSION 3.25)
project(MmdPhysicsConsumerProbe LANGUAGES CXX)

find_package(physicsUsd 0.1 CONFIG REQUIRED)
if(NOT physicsUsd_PACKAGE_CONTRACT_VERSION EQUAL 1)
  message(FATAL_ERROR "MMD probe requires physics packages with descriptor/build compatibility metadata; rebuild shared packages into a clean prefix")
endif()
if(NOT PXR_VERSION STREQUAL "2608")
  message(FATAL_ERROR "MMD physics probe requires OpenUSD 26.08 exactly")
endif()
add_executable(mmd_physics_consumer_probe main.cpp)
target_compile_features(mmd_physics_consumer_probe PRIVATE cxx_std_17)
target_link_libraries(mmd_physics_consumer_probe PRIVATE physicsUsd::physicsUsd usdSkel)
option(MMD_PHYSICS_EXECUTION "Exercise the installed optional six-DOF backend" OFF)
option(MMD_PHYSICS_TRANSLATION_SPRINGS "Approximate positive MMD translation springs as Hookean springs" OFF)
option(MMD_PHYSICS_ROTATION_SPRINGS "Approximate positive MMD rotation springs with small-angle motors" OFF)
if(MMD_PHYSICS_ROTATION_SPRINGS AND NOT MMD_PHYSICS_EXECUTION)
  message(FATAL_ERROR "Rotation springs require MMD_PHYSICS_EXECUTION=ON")
endif()
if(MMD_PHYSICS_ROTATION_SPRINGS)
  target_compile_definitions(mmd_physics_consumer_probe PRIVATE MMD_PHYSICS_ROTATION_SPRINGS)
endif()
if(MMD_PHYSICS_TRANSLATION_SPRINGS AND NOT MMD_PHYSICS_EXECUTION)
  message(FATAL_ERROR "Translation springs require MMD_PHYSICS_EXECUTION=ON")
endif()
if(MMD_PHYSICS_TRANSLATION_SPRINGS)
  target_compile_definitions(mmd_physics_consumer_probe PRIVATE MMD_PHYSICS_TRANSLATION_SPRINGS)
endif()
if(MMD_PHYSICS_EXECUTION)
  find_package(physicsJolt 0.1 CONFIG REQUIRED)
  if(NOT physicsJolt_PACKAGE_CONTRACT_VERSION EQUAL 1)
    message(FATAL_ERROR "MMD execution requires rebuilt physicsJolt compatibility metadata")
  endif()
  if(NOT physicsJolt_SIX_DOF_AVAILABLE)
    message(FATAL_ERROR "Execution probe requires physicsJolt built with PHYSICSJOLT_ENABLE_SIX_DOF=ON")
  endif()
  target_link_libraries(mmd_physics_consumer_probe PRIVATE physicsJolt::physicsJolt)
  target_compile_definitions(mmd_physics_consumer_probe PRIVATE MMD_PHYSICS_EXECUTION)
endif()

include(CTest)
set(MMD_PHYSICS_MMD_PREFIX "" CACHE PATH "Installed MMD importer/schema prefix for PMX regression tests")
option(MMD_PHYSICS_FULL_CORPUS "Run every opening generated PMX instead of the bounded representative set" OFF)
if(MMD_PHYSICS_FULL_CORPUS AND NOT MMD_PHYSICS_MMD_PREFIX)
  message(FATAL_ERROR "Full corpus requires MMD_PHYSICS_MMD_PREFIX")
endif()
if(BUILD_TESTING)
  add_executable(mmd_physics_declarations_test stage_declarations_test.cpp)
  target_compile_features(mmd_physics_declarations_test PRIVATE cxx_std_17)
  target_link_libraries(mmd_physics_declarations_test PRIVATE physicsUsd::physicsUsd usdSkel)
  add_test(NAME mmd_physics_stage_declarations COMMAND mmd_physics_declarations_test)
  add_executable(mmd_physics_binding_test bone_binding_test.cpp)
  target_compile_features(mmd_physics_binding_test PRIVATE cxx_std_17)
  target_link_libraries(mmd_physics_binding_test PRIVATE physicsUsd::physicsUsd usdSkel)
  add_test(NAME mmd_physics_bone_binding COMMAND mmd_physics_binding_test)
  add_test(NAME mmd_physics_bone_binding_probe COMMAND "${CMAKE_COMMAND}"
    "-DPROBE=$<TARGET_FILE:mmd_physics_consumer_probe>"
    "-DWORK_DIR=${CMAKE_CURRENT_BINARY_DIR}/bone-binding-probe"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/bone_binding_probe_test.cmake")
  if(MMD_PHYSICS_MMD_PREFIX)
    foreach(bundle mmdSchema usdMmdFileFormat)
      if(NOT EXISTS "${MMD_PHYSICS_MMD_PREFIX}/plugin/resources/${bundle}/plugInfo.json")
        message(FATAL_ERROR "MMD fixture tests require installed ${bundle} in MMD_PHYSICS_MMD_PREFIX")
      endif()
    endforeach()
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    set(fixture_args)
    if(MMD_PHYSICS_FULL_CORPUS)
      list(APPEND fixture_args --all)
    endif()
    if(MMD_PHYSICS_EXECUTION)
      list(APPEND fixture_args --execution)
    endif()
    if(MMD_PHYSICS_TRANSLATION_SPRINGS)
      list(APPEND fixture_args --translation-springs)
    endif()
    if(MMD_PHYSICS_ROTATION_SPRINGS)
      list(APPEND fixture_args --rotation-springs)
    endif()
    add_test(NAME mmd_physics_pmx_fixtures COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/check_fixtures.py"
      --probe "$<TARGET_FILE:mmd_physics_consumer_probe>"
      --fixtures "${CMAKE_CURRENT_SOURCE_DIR}/../../plugins/usdMmdFileFormat/tests/fixtures"
      ${fixture_args})
    set_tests_properties(mmd_physics_pmx_fixtures PROPERTIES TIMEOUT 150 LABELS "integration")
    set_property(TEST mmd_physics_pmx_fixtures PROPERTY ENVIRONMENT_MODIFICATION
      "PXR_PLUGINPATH_NAME=set:"
      "PXR_PLUGINPATH_NAME=path_list_append:${MMD_PHYSICS_MMD_PREFIX}/plugin/resources/mmdSchema"
      "PXR_PLUGINPATH_NAME=path_list_append:${MMD_PHYSICS_MMD_PREFIX}/plugin/resources/usdMmdFileFormat"
      "PATH=path_list_prepend:${MMD_PHYSICS_MMD_PREFIX}/lib")
    add_test(NAME mmd_physics_fixture_checker COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/test_check_fixtures.py")
  endif()
  if(MMD_PHYSICS_ROTATION_SPRINGS)
    add_executable(mmd_physics_rotation_springs_test rotation_springs_test.cpp)
    target_compile_features(mmd_physics_rotation_springs_test PRIVATE cxx_std_17)
    target_link_libraries(mmd_physics_rotation_springs_test PRIVATE physicsUsd::physicsUsd physicsJolt::physicsJolt)
    add_test(NAME mmd_physics_rotation_springs COMMAND mmd_physics_rotation_springs_test)
    add_test(NAME mmd_physics_rotation_springs_probe COMMAND "${CMAKE_COMMAND}"
      "-DPROBE=$<TARGET_FILE:mmd_physics_consumer_probe>"
      "-DWORK_DIR=${CMAKE_CURRENT_BINARY_DIR}/rotation-springs-probe"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/rotation_springs_probe_test.cmake")
  endif()
  if(MMD_PHYSICS_TRANSLATION_SPRINGS)
    add_executable(mmd_physics_translation_springs_test translation_springs_test.cpp)
    target_compile_features(mmd_physics_translation_springs_test PRIVATE cxx_std_17)
    target_link_libraries(mmd_physics_translation_springs_test PRIVATE physicsUsd::physicsUsd physicsJolt::physicsJolt)
    add_test(NAME mmd_physics_translation_springs COMMAND mmd_physics_translation_springs_test)
    add_test(NAME mmd_physics_translation_springs_probe COMMAND "${CMAKE_COMMAND}"
      "-DPROBE=$<TARGET_FILE:mmd_physics_consumer_probe>"
      "-DWORK_DIR=${CMAKE_CURRENT_BINARY_DIR}/translation-springs-probe"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/translation_springs_probe_test.cmake")
  endif()
  add_executable(mmd_physics_filter_test collision_filter_test.cpp)
  target_compile_features(mmd_physics_filter_test PRIVATE cxx_std_17)
  target_link_libraries(mmd_physics_filter_test PRIVATE physicsUsd::physicsUsd)
  if(MMD_PHYSICS_EXECUTION)
    target_link_libraries(mmd_physics_filter_test PRIVATE physicsJolt::physicsJolt)
    target_compile_definitions(mmd_physics_filter_test PRIVATE MMD_PHYSICS_EXECUTION)
  endif()
  add_test(NAME mmd_physics_collision_filter COMMAND mmd_physics_filter_test)
  add_executable(mmd_physics_material_test contact_material_test.cpp)
  target_compile_features(mmd_physics_material_test PRIVATE cxx_std_17)
  target_link_libraries(mmd_physics_material_test PRIVATE physicsUsd::physicsUsd)
  if(MMD_PHYSICS_EXECUTION)
    target_link_libraries(mmd_physics_material_test PRIVATE physicsJolt::physicsJolt)
    target_compile_definitions(mmd_physics_material_test PRIVATE MMD_PHYSICS_EXECUTION)
  endif()
  add_test(NAME mmd_physics_contact_material COMMAND mmd_physics_material_test)
  add_executable(mmd_physics_damping_test body_damping_test.cpp)
  target_compile_features(mmd_physics_damping_test PRIVATE cxx_std_17)
  target_link_libraries(mmd_physics_damping_test PRIVATE physicsUsd::physicsUsd)
  if(MMD_PHYSICS_EXECUTION)
    target_link_libraries(mmd_physics_damping_test PRIVATE physicsJolt::physicsJolt)
    target_compile_definitions(mmd_physics_damping_test PRIVATE MMD_PHYSICS_EXECUTION)
  endif()
  add_test(NAME mmd_physics_body_damping COMMAND mmd_physics_damping_test)
  add_executable(mmd_physics_springs_test joint_springs_test.cpp)
  target_compile_features(mmd_physics_springs_test PRIVATE cxx_std_17)
  target_link_libraries(mmd_physics_springs_test PRIVATE physicsUsd::physicsUsd)
  add_test(NAME mmd_physics_joint_springs COMMAND mmd_physics_springs_test)
  add_test(NAME mmd_physics_joint_springs_probe COMMAND "${CMAKE_COMMAND}"
    "-DPROBE=$<TARGET_FILE:mmd_physics_consumer_probe>"
    "-DWORK_DIR=${CMAKE_CURRENT_BINARY_DIR}/joint-springs-probe"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/joint_springs_probe_test.cmake")
endif()
